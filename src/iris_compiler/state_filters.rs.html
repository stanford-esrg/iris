16pub(crate) fn gen_state_filters(
17    sub: &SubscriptionDecoder,
18    statics: &mut BTreeMap<String, (String, proc_macro2::TokenStream)>,
19) -> (proc_macro2::TokenStream, proc_macro2::TokenStream) {
20    let mut fns = vec![];
21    let mut main = vec![];
22    for tx in StateTransition::iter() {
23        if tx == StateTransition::Packet {
24            continue;
25        }
26        if !sub.requires_filter(&tx) {
27            continue;
28        }
29        let ptree = sub.build_ptree(tx);
30        let mut body: Vec<proc_macro2::TokenStream> = vec![];
31
32        // Ensure root delivery/matches are covered
33        if !ptree.root.deliver.is_empty() || !ptree.root.actions.drop() {
34            update_body(&mut body, &ptree.root, sub);
35        }
36        let extract_sessions = matches!(
37            tx.compare(&StateTransition::L7EndHdrs),
38            StateTxOrd::Greater | StateTxOrd::Equal
39        );
40        gen_state_filter_util(
41            &mut body,
42            &ptree.root,
43            &ptree,
44            statics,
45            sub,
46            extract_sessions,
47        );
48        let fn_name = Ident::new(&(format!("tx_{}", tx).to_lowercase()), Span::call_site());
49
50        let ident = Ident::new(&tx.to_string(), Span::call_site());
51        main.push(quote! {
52            iris_core::StateTransition::#ident => #fn_name(conn, &tx, pdu),
53        });
54
55        // Ensure that datatypes and custom filters that requested updates
56        // at this state transition receive them.
57        let mut update = quote! {};
58        if !tx.is_streaming() {
59            update = update_to_tokens(sub, &tx);
60            if !update.is_empty() {
61                update = quote! {
62                    #update
63                };
64            }
65        }
66
67        // Start/end for filtered/tracked data
68        let mut start_tracked = quote! {};
69        let mut end_tracked = quote! {};
70        for dt in ptree.filtered_datatypes {
71            let dt_ident = Ident::new(&dt.to_lowercase(), Span::call_site());
72            start_tracked = quote! {
73                #start_tracked
74                conn.tracked.#dt_ident.start_state_tx(&tx);
75            };
76            end_tracked = quote! {
77                #end_tracked
78                conn.tracked.#dt_ident.end_state_tx();
79            };
80        }
81
82        // Complete state transition handler
83        let extract_data = if matches!(tx, StateTransition::L4FirstPacket) {
84            quote! { let pdu = pdu.expect("L4Pdu not passed to L4FirstPacket?"); }
85        } else {
86            quote! {}
87        };
88        fns.push(quote! {
89            fn #fn_name(conn: &mut ConnInfo<TrackedWrapper>, tx: &iris_core::StateTransition, pdu: Option<&iris_core::L4Pdu>) {
90                let mut ret = false; // unused in state_tx filters
91                #extract_data
92                #start_tracked
93                let tx_data = iris_core::StateTxData::from_tx(tx, &conn.layers[0]);
94                // Some callbacks/filters may require immutable borrow of `conn`;
95                // it's easiest to just limit the body of the function to immutable
96                // borrows and then update actions at the end.
97                let mut transport_actions = iris_core::conntrack::TrackedActions::new();
98                let mut layer0_actions = iris_core::conntrack::TrackedActions::new();
99                // Update filters, datatypes first
100                #update
101                #( #body )*
102                conn.linfo.actions.extend(&transport_actions);
103                conn.layers[0].extend_actions(&layer0_actions);
104                #end_tracked
105            }
106        });
107    }
108    (
109        quote! {
110            match tx {
111                #( #main )*
112                _ => { },
113            }
114        },
115        quote! { #( #fns )* },
116    )
117}

119fn gen_state_filter_util(
120    code: &mut Vec<proc_macro2::TokenStream>,
121    node: &PNode,
122    tree: &PTree,
123    statics: &mut BTreeMap<String, (String, proc_macro2::TokenStream)>,
124    sub: &SubscriptionDecoder,
125    extract_sessions: bool,
126) {
127    let mut first_unary = true;
128    for child in &node.children {
129        match &child.pred {
130            Predicate::Unary { protocol } => {
131                if child.pred.on_packet() {
132                    add_unary_pred(
133                        code,
134                        child,
135                        tree,
136                        protocol,
137                        statics,
138                        first_unary,
139                        sub,
140                        extract_sessions,
141                    );
142                    first_unary = false;
143                } else if child.pred.on_proto() {
144                    add_service_pred(code, child, tree, protocol, statics, sub, extract_sessions);
145                } else {
146                    panic!("Unknown unary predicate: {}", child.pred);
147                }
148            }
149            Predicate::Binary {
150                protocol,
151                field,
152                op,
153                value,
154            } => {
155                if child.pred.on_packet() || child.pred.on_session() {
156                    let pred_tokenstream = binary_to_tokens(protocol, field, op, value, statics);
157                    add_pred(
158                        code,
159                        child,
160                        tree,
161                        pred_tokenstream,
162                        statics,
163                        sub,
164                        extract_sessions,
165                    );
166                } else {
167                    panic!("Unknown binary predicate: {}", child.pred);
168                }
169            }
170            Predicate::LayerState { layer, state, op } => {
171                let extract_sessions_ = extract_sessions
172                    || (layer == &SupportedLayer::L7 && state >= &LayerState::Payload);
173                let pred_tokenstream = layerstate_to_tokens(layer, state, *op);
174                add_pred(
175                    code,
176                    child,
177                    tree,
178                    pred_tokenstream,
179                    statics,
180                    sub,
181                    extract_sessions_,
182                );
183            }
184            Predicate::Custom { name, matched, .. } => {
185                let pred_tokenstream = custom_pred_to_tokens(&name.0, *matched, sub);
186                add_pred(
187                    code,
188                    child,
189                    tree,
190                    pred_tokenstream,
191                    statics,
192                    sub,
193                    extract_sessions,
194                );
195            }
196            Predicate::Callback { name } => {
197                assert!(
198                    child.children.is_empty()
199                        || (
200                            // In general, a callback node shouldn't have any other children.
201                            // One exception: it can be OK to have a State predicate (e.g., "if L7 >= Headers") here if
202                            // a filter has already terminated/matched, but the callback is still waiting for
203                            // a data type to be constructed (e.g., a TLS handshake, but we're still in "L7Headers");
204                            // here, the callback is "active" but not yet invoked.
205                            // This is a sanity-check for these cases.
206                            // Possible opportunity for REFACTOR of PTrees in the future.
207                            child.children.iter().all(|c| c.pred.is_state()
208                                && c.children.is_empty()
209                                && !c.actions.drop())
210                        ),
211                    "Expect callback predicate {} to terminate pattern; found children: {}",
212                    child.pred,
213                    child
214                        .children
215                        .iter()
216                        .map(|c| c.pred.to_string())
217                        .collect::<Vec<_>>()
218                        .join("\n")
219                );
220                add_callback_pred(code, &name.0, child, tree, statics, sub, extract_sessions);
221            }
222        }
223    }
224}

226#[allow(clippy::too_many_arguments)]
227fn add_unary_pred(
228    code: &mut Vec<proc_macro2::TokenStream>,
229    node: &PNode,
230    tree: &PTree,
231    protocol: &ProtocolName,
232    statics: &mut BTreeMap<String, (String, proc_macro2::TokenStream)>,
233    first_unary: bool,
234    sub: &SubscriptionDecoder,
235    extract_sessions: bool,
236) {
237    let ident = Ident::new(protocol.name(), Span::call_site());
238    let ident_type = Ident::new(
239        &(protocol.name().to_owned().to_camel_case() + "CData"),
240        Span::call_site(),
241    );
242    let pred_tokenstream = quote! {
243        &iris_core::protocols::stream::ConnData::parse_to::<iris_core::protocols::stream::conn::#ident_type>(&conn.cdata)
244    };
245
246    let mut body: Vec<proc_macro2::TokenStream> = vec![];
247    update_body(&mut body, node, sub);
248    gen_state_filter_util(&mut body, node, tree, statics, sub, extract_sessions);
249
250    if first_unary {
251        code.push(quote! {
252            if let Ok(#ident) = #pred_tokenstream {
253                #( #body )*
254            }
255        });
256    } else {
257        code.push(quote! {
258            else if let Ok(#ident) = #pred_tokenstream {
259                #( #body )*
260            }
261        });
262    }
263}

265#[allow(clippy::too_many_arguments)]
266fn add_service_pred(
267    code: &mut Vec<proc_macro2::TokenStream>,
268    node: &PNode,
269    tree: &PTree,
270    protocol: &ProtocolName,
271    statics: &mut BTreeMap<String, (String, proc_macro2::TokenStream)>,
272    sub: &SubscriptionDecoder,
273    extract_sessions: bool,
274) {
275    let service_ident = Ident::new(&protocol.name().to_camel_case(), Span::call_site());
276    // Destructuring `last_session()` both binds the session variable and tests that a
277    // session was actually parsed. Only do it when some descendant predicate needs the
278    // binding: `binary_to_tokens` emits `#proto.#field()` against it.
279    //
280    // Otherwise this is a protocol check.
281    // The compiler grants L7 `Parse` past `L7OnDisc` only when a subscription needs the
282    // session, so, for a protocol-only pattern, parsing stops as soon as the protocol is
283    // known and `last_session()` stays empty for the rest of the connection.
284    let pred_tokenstream = if extract_sessions && binds_session(node, protocol) {
285        let proto_ident = Ident::new(protocol.name(), Span::call_site());
286        quote! {
287            let iris_core::protocols::stream::SessionData::#service_ident(#proto_ident) = &conn.layers[0].last_session().data
288        }
289    } else {
290        quote! {
291            matches!(conn.layers[0].last_protocol(), iris_core::protocols::stream::SessionProto::#service_ident)
292        }
293    };
294    add_pred(
295        code,
296        node,
297        tree,
298        pred_tokenstream,
299        statics,
300        sub,
301        extract_sessions,
302    );
303}

305/// Returns `true` if any descendant of `node` reads a session field of `protocol`, and so
306/// needs the session variable that `add_service_pred`'s destructuring form binds.
307fn binds_session(node: &PNode, protocol: &ProtocolName) -> bool {
308    node.children.iter().any(|c| {
309        (c.pred.on_session() && c.pred.get_protocol() == protocol) || binds_session(c, protocol)
310    })
311}

313fn add_pred(
314    code: &mut Vec<proc_macro2::TokenStream>,
315    node: &PNode,
316    tree: &PTree,
317    pred_tokenstream: proc_macro2::TokenStream,
318    statics: &mut BTreeMap<String, (String, proc_macro2::TokenStream)>,
319    sub: &SubscriptionDecoder,
320    extract_sessions: bool,
321) {
322    let mut body: Vec<proc_macro2::TokenStream> = vec![];
323    update_body(&mut body, node, sub);
324    gen_state_filter_util(&mut body, node, tree, statics, sub, extract_sessions);
325    if node.if_else {
326        code.push(quote! {
327            else if #pred_tokenstream {
328                #( #body )*
329            }
330        });
331    } else {
332        code.push(quote! {
333            if #pred_tokenstream {
334                #( #body )*
335            }
336        });
337    }
338}

340#[allow(clippy::too_many_arguments)]
341fn add_callback_pred(
342    code: &mut Vec<proc_macro2::TokenStream>,
343    name: &String,
344    node: &PNode,
345    tree: &PTree,
346    statics: &mut BTreeMap<String, (String, proc_macro2::TokenStream)>,
347    sub: &SubscriptionDecoder,
348    extract_sessions: bool,
349) {
350    // If we're at the callback predicate, then the CB is ready to
351    // be invoked or set as active (if it hasn't already unsubscribed).
352    for deliver in &node.deliver {
353        assert!(
354            &deliver.subscription_id == name,
355            "Found callback {} at {} callback pred node",
356            deliver.subscription_id,
357            name
358        );
359        let cb = fil_callback_to_tokens(sub, deliver, Some(node));
360        code.push(quote! { #cb });
361    }
362
363    // Actions conditioned on whether callback is active
364    let pred_tokenstream = callback_pred_to_tokens(name);
365    let mut body = vec![];
366    if !node.actions.drop() {
367        let actions = data_actions_to_tokens(&node.actions);
368        body.push(quote! { #actions });
369    }
370    // A callback node can carry LayerState children if the callback filter matched but
371    // the function(s) are still waiting on a datatype.
372    gen_state_filter_util(&mut body, node, tree, statics, sub, extract_sessions);
373    code.push(quote! {
374        if #pred_tokenstream {
375            #( #body )*
376        }
377    });
378}

380fn update_body(body: &mut Vec<proc_macro2::TokenStream>, node: &PNode, sub: &SubscriptionDecoder) {
381    if !node.actions.drop() {
382        let actions = data_actions_to_tokens(&node.actions);
383        body.push(quote! { #actions });
384    }
385    for matched in &node.matched {
386        // Note: setting `matched` to active must come before `deliver`
387        let cb = cb_set_active_to_tokens(matched);
388        body.push(quote! { #cb });
389    }
390    for deliver in &node.deliver {
391        let cb = fil_callback_to_tokens(sub, deliver, Some(node));
392        body.push(quote! { #cb });
393    }
394    for dt in node.filtered_datatypes.values() {
395        let dt = filtered_dt_to_tokens(dt);
396        body.push(quote! { #dt });
397    }
398}

436    fn callback(name: &str, filter: &str, datatype: &str) -> String {
437        let func = format!(
438            r#"{{"name":"{}","datatypes":["{}"],"returns":"None"}}"#,
439            name, datatype
440        );
441        format!(
442            r#"{{"Callback":{{"filter":"{}","level":["L4Terminated"],"func":{},"expl_parsers":[]}}}}"#,
443            filter, func

447    /// Runs the real codegen path over `defs` and returns the generated state-transition
448    /// functions as a token string.
449    fn gen_fns(defs: &[&str]) -> String {
450        let inputs: Vec<ParsedInput> = defs
451            .iter()
452            .flat_map(|d| d.lines())
453            .map(str::trim)
454            .filter(|l| !l.is_empty())
455            .map(|l| serde_json::from_str(l).unwrap_or_else(|e| panic!("{}: {}", e, l)))
456            .collect();
457        let sub = SubscriptionDecoder::new(&inputs);
458        let mut statics = BTreeMap::new();
459        gen_state_filters(&sub, &mut statics).1.to_string()
460    }

462    /// Extracts the body of the generated `tx_<state>` function from `code`.
463    fn tx_fn(code: &str, state: &str) -> String {
464        let marker = format!("fn tx_{} ", state);
465        let start = code
466            .find(&marker)
467            .unwrap_or_else(|| panic!("no tx_{} in generated code:\n{}", state, code));
468        let rest = &code[start + marker.len()..];
469        // Functions are emitted back to back, so the next `fn tx_` ends this one.
470        match rest.find("fn tx_") {
471            Some(end) => rest[..end].to_string(),
472            None => rest.to_string(),
473        }
474    }

476    /// A protocol-only pattern -- one naming an L7 protocol but reading no session field,
477    /// whose callback wants no session-derived datatype -- must be dispatched on the
478    /// identified protocol, never on the presence of a parsed session.
479    ///
480    /// The action planner keeps L7 `Parse` alive past `L7OnDisc` only when a subscription
481    /// needs the session. A protocol-only pattern does not, so parsing stops the moment
482    /// the protocol is known and `last_session()` stays empty for the rest of the
483    /// connection. Testing `SessionData::<Proto>` there is unsatisfiable forever.
484    ///
485    /// The predicate only reaches `L7EndHdrs`/`L4Terminated` at all when something else in
486    /// the binary keeps non-matching connections alive that far -- otherwise arriving there
487    /// already implies the match and the predicate is pruned. Each case below is one way
488    /// for that to happen; none of them is special, which is why this is easy to hit.
489    #[test]
490    fn protocol_only_pattern_dispatches_on_protocol_not_session() {
491        let cases: Vec<(&str, Vec<String>)> = vec![
492            // A second protocol callback: neither predicate implies the other.
493            (
494                "two protocols",
495                vec![conn_callback("a", "tls"), conn_callback("b", "ssh")],
496            ),
497            // A transport-level callback: every connection survives to L4Terminated.
498            (
499                "protocol + transport",
500                vec![conn_callback("a", "tls"), conn_callback("b", "udp")],
501            ),
502            // A custom filter that resolves no earlier than L7OnDisc.
503            (
504                "protocol + streaming filter",
505                vec![
506                    conn_callback("a", "tls"),
507                    conn_callback("b", "StreamFilter"),
508                ],
509            ),
510            // A disjunction, which leaves both arms to be told apart at dispatch.
511            (
512                "disjunction",
513                vec![conn_callback("a", "tls or ssh"), conn_callback("b", "udp")],
514            ),
515        ];
516
517        for (name, defs) in cases {
518            let mut all = vec![CONN_DATATYPE.to_string(), STREAM_FILTER.to_string()];
519            all.extend(defs);
520            let refs: Vec<&str> = all.iter().map(String::as_str).collect();
521            let term = tx_fn(&gen_fns(&refs), "l4terminated");
522            assert!(
523                term.contains("last_protocol"),
524                "[{}] should dispatch on last_protocol() at L4Terminated:\n{}",
525                name,
526                term
527            );
528            assert!(
529                !term.contains("SessionData"),
530                "[{}] no session field is read, so dispatch must not require a parsed \
531                 session -- for a protocol-only pattern one is never produced, so this \
532                 would never match:\n{}",
533                name,
534                term
535            );
536        }
537    }

1//! A streaming filter that heuristically accepts connections that look like
2//! Zoom: on a UDP port in [`ZOOM_PORT_RANGE`], with initial packets whose
3//! first byte matches one of Zoom's observed payload-type markers.
4
5#[allow(unused_imports)]
6use iris_compiler::{filter, filter_fn};
7use iris_core::protocols::packet::udp::UDP_PROTOCOL;
8use iris_core::subscription::{FilterResult, StreamingFilter};
9use iris_core::L4Pdu;
10
11/// Number of payload-bearing packets inspected before making a decision.
12pub const MAYBE_ZOOM_WINDOW: usize = 30;
13/// Fraction of the first [`MAYBE_ZOOM_WINDOW`] payload-bearing packets that
14/// must look like Zoom for the connection to be accepted.
15pub const MAYBE_ZOOM_MIN_FRACTION: f64 = 0.95;
16/// Minimum payload-bearing packets needed to judge a connection that ended
17/// before [`MAYBE_ZOOM_WINDOW`] was reached.
18pub const MAYBE_ZOOM_MIN_PKTS: usize = MAYBE_ZOOM_WINDOW;
19/// UDP port range Zoom traffic is expected on.
20pub const ZOOM_PORT_RANGE: std::ops::RangeInclusive<u16> = 8801..=8810;
21/// First-byte values that mark a Zoom payload.
22pub const ZOOM_FIRST_BYTES: [u8; 4] = [0x03, 0x04, 0x05, 0x07];
23
24/// Check that the first byte matches one of [`ZOOM_FIRST_BYTES`].
25#[inline]
26fn is_zoom_payload(payload: &[u8]) -> bool {
27    matches!(payload.first(), Some(b) if ZOOM_FIRST_BYTES.contains(b))
28}
29
30/// Number of matching payloads required within a full window of
31/// [`MAYBE_ZOOM_WINDOW`] packets to meet [`MAYBE_ZOOM_MIN_FRACTION`].
32#[inline]
33fn required_matches() -> usize {
34    (MAYBE_ZOOM_WINDOW as f64 * MAYBE_ZOOM_MIN_FRACTION).ceil() as usize
35}
36
37/// Accepts a connection once at least [`MAYBE_ZOOM_MIN_FRACTION`] of its
38/// first [`MAYBE_ZOOM_WINDOW`] payload-bearing packets (UDP, port in
39/// [`ZOOM_PORT_RANGE`]) start with one of [`ZOOM_FIRST_BYTES`]. Non-UDP or
40/// non-Zoom-port connections are dropped immediately.
41#[cfg_attr(not(feature = "skip_expand"), filter)]
42#[derive(Debug)]
43pub struct MaybeZoom {
44    /// Payload-bearing packets inspected so far.
45    seen: usize,
46    /// Of those, how many matched a Zoom first-byte marker.
47    matched: usize,
48}
49
50impl StreamingFilter for MaybeZoom {
51    fn new(_first_packet: &L4Pdu) -> Self {
52        Self {
53            seen: 0,
54            matched: 0,
55        }
56    }
57
58    fn clear(&mut self) {
59        self.seen = 0;
60        self.matched = 0;
61    }
62}
63
64impl MaybeZoom {
65    /// Updates counts from one payload
66    fn record(&mut self, data: &[u8]) {
67        self.seen += 1;
68        if is_zoom_payload(data) {
69            self.matched += 1;
70        }
71    }
72
73    /// Resolves the current counts to a `FilterResult`
74    fn decide(&self) -> FilterResult {
75        let required = required_matches();
76        if self.matched >= required {
77            return FilterResult::Accept;
78        }
79        if self.seen >= MAYBE_ZOOM_WINDOW {
80            // Window exhausted
81            return FilterResult::Drop;
82        }
83        // Impossible to reach required number of matches
84        let remaining = MAYBE_ZOOM_WINDOW - self.seen;
85        if self.matched + remaining < required {
86            return FilterResult::Drop;
87        }
88        // Not enough evidence yet
89        FilterResult::Continue
90    }
91
92    #[cfg_attr(not(feature = "skip_expand"), filter_fn("MaybeZoom,level=InL4Conn"))]
93    pub fn update(&mut self, pdu: &L4Pdu) -> FilterResult {
94        // Drop traffic not on a UDP port in ZOOM_PORT_RANGE
95        if pdu.ctxt.proto != UDP_PROTOCOL
96            || (!ZOOM_PORT_RANGE.contains(&pdu.ctxt.src.port())
97                && !ZOOM_PORT_RANGE.contains(&pdu.ctxt.dst.port()))
98        {
99            return FilterResult::Drop;
100        }
101        // Skip packets with no payload
102        if pdu.length() == 0 {
103            return FilterResult::Continue;
104        }
105        // Extract data
106        if let Ok(data) = pdu.mbuf_ref().get_data_slice(pdu.offset(), pdu.length()) {
107            self.record(data);
108        }
109        self.decide()
110    }
111
112    /// Reached only if the connection terminated before the window filled.
113    /// The fraction is applied to the packets actually observed, but only
114    /// once there are enough of them to be meaningful.
115    #[cfg_attr(
116        not(feature = "skip_expand"),
117        filter_fn("MaybeZoom,level=L4Terminated")
118    )]
119    pub fn terminated(&self) -> FilterResult {
120        if self.seen >= MAYBE_ZOOM_MIN_PKTS
121            && self.matched as f64 >= self.seen as f64 * MAYBE_ZOOM_MIN_FRACTION
122        {
123            FilterResult::Accept
124        } else {
125            FilterResult::Drop
126        }
127    }
128}

130#[cfg(test)]
131mod tests {
132    use super::*;
133
134    #[test]
135    fn zoom_payload_bit_pattern() {
136        assert!(is_zoom_payload(&[0x03, 0x00]));
137        assert!(is_zoom_payload(&[0x04]));
138        assert!(is_zoom_payload(&[0x05]));
139        assert!(is_zoom_payload(&[0x07]));
140        assert!(!is_zoom_payload(&[0x00]));
141        assert!(!is_zoom_payload(&[0x06]));
142        assert!(!is_zoom_payload(&[]));
143    }
144
145    #[test]
146    fn required_matches_matches_configured_threshold() {
147        // 30 * 0.95 = 28.5, so 29 of 30 -- tolerates exactly one stray packet.
148        assert_eq!(required_matches(), 29);
149    }
150
151    #[test]
152    fn terminated_drops_flows_below_the_evidence_floor() {
153        let mut f = MaybeZoom::new_for_test();
154        for b in [0x03, 0x04]
155            .into_iter()
156            .cycle()
157            .take(MAYBE_ZOOM_MIN_PKTS - 1)
158        {
159            f.record(&[b]);
160        }
161        assert!(matches!(f.terminated(), FilterResult::Drop));
162
163        let mut f = MaybeZoom::new_for_test();
164        for b in [0x03, 0x04].into_iter().cycle().take(MAYBE_ZOOM_MIN_PKTS) {
165            f.record(&[b]);
166        }
167        assert!(matches!(f.terminated(), FilterResult::Accept));
168    }
169
170    #[test]
171    fn terminated_applies_fraction_to_observed_packets() {
172        let mut f = MaybeZoom::new_for_test();
173        for _ in 0..29 {
174            f.record(&[0x03]);
175        }
176        for _ in 0..1 {
177            f.record(&[0x00]); // not a Zoom marker
178        }
179        // 30 seen, 29 matched -- 97% >= 95%
180        assert_eq!(f.seen, 30);
181        assert!(matches!(f.terminated(), FilterResult::Accept));
182
183        let mut f = MaybeZoom::new_for_test();
184        for _ in 0..27 {
185            f.record(&[0x03]);
186        }
187        for _ in 0..3 {
188            f.record(&[0x00]);
189        }
190        // 30 seen, 27 matched -- 90% < 95%
191        assert!(matches!(f.terminated(), FilterResult::Drop));
192    }
193
194    #[test]
195    fn decide_drops_once_window_exhausted_below_threshold() {
196        let mut f = MaybeZoom::new_for_test();
197        let mut result = FilterResult::Continue;
198        for i in 0..MAYBE_ZOOM_WINDOW {
199            if i < 9 {
200                f.record(&[0x03]);
201            } else {
202                f.record(&[0x00]);
203            }
204            result = f.decide();
205        }
206        assert!(matches!(result, FilterResult::Drop));
207    }
208
209    #[test]
210    fn decide_accepts_once_threshold_reached() {
211        let mut f = MaybeZoom::new_for_test();
212        for _ in 0..(required_matches() - 1) {
213            f.record(&[0x04]);
214        }
215        assert!(matches!(f.decide(), FilterResult::Continue));
216        f.record(&[0x05]);
217        assert!(matches!(f.decide(), FilterResult::Accept));
218    }
219
220    impl MaybeZoom {
221        fn new_for_test() -> Self {
222            Self {
223                seen: 0,
224                matched: 0,
225            }
226        }
227    }

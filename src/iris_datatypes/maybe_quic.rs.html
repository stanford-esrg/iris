1//! A streaming filter that heuristically accepts connections that look like
2//! QUIC: on an expected port, not clearly another protocol, and initial
3//! packets with (potential) QUIC short headers.
4//!
5//! Per RFC 9000, the first byte of a QUIC short header starts with bits
6//! `01`: header form 0, fixed bit 1. Header protection (Section 5.4.1) masks
7//! only the low five bits of that byte, so the top two are directly
8//! observable and the low five are pseudorandom per packet -- a genuine
9//! 32-value uniform source, which the entropy check below relies on.
10//!
11//! Three shapes count as evidence of QUIC (see [`Shape`]):
12//!
13//! - **Short header** -- `01` in the top two bits.
14//! - **Long header** -- high bit set, a version Iris recognizes, and not
15//!   Version Negotiation (`0x00000000`, which is no evidence the connection
16//!   will carry a 1-RTT packet, and a shape length- and ID-prefixed binary
17//!   formats hit by accident). Meant to capture connections observed
18//!   mid-handshake and coalesced datagrams.
19//! - **Greased short header** -- an endpoint whose peer advertised
20//!   `grease_quic_bit` may set the fixed bit to any value on its 1-RTT
21//!   packets (RFC 9287), so roughly half of a greasing endpoint's packets
22//!   have it clear. Credited only once the connection has shown at least one
23//!   unambiguous QUIC packet, and only when the datagram is large enough to
24//!   hold a 1-RTT packet.
25//!
26//! For a datagram at least [`MIN_1RTT_DATAGRAM_LEN`] bytes, the short-header
27//! and greased shapes together cover every value with the high bit clear --
28//! the fixed bit only sorts a packet into one bucket or the other, and both
29//! count toward acceptance. What actually filters non-QUIC traffic is:
30//! requiring at least one unambiguous packet before anything is credited;
31//! requiring [`MAYBE_QUIC_MIN_DISTINCT_LOW5`] distinct low-5-bit values
32//! across the matched packets -- or one recognized long header in place of
33//! that, since a 32-bit version match is stronger on its own -- which is
34//! nearly free against traffic with a uniformly random first byte and
35//! rejects protocols that pin their low bits (TURN ChannelData, uTP,
36//! OpenVPN); and the window, fraction, and packet floor below, which set the
37//! false-positive rate against that random-looking traffic directly.
38//!
39//! Greased packets are counted toward the entropy check: RFC 9287 greasing
40//! flips only the fixed bit, never the low five, so a greased packet carries
41//! the same per-packet entropy as any other 1-RTT packet. Masking to five
42//! bits is what keeps that safe -- a TURN relay mixing `0x40`/`0x41`
43//! ChannelData with `0x00`/`0x01` STUN collapses to two distinct low-5
44//! values either way, still short of the gate.

54/// Number of payload-bearing packets inspected before making a decision.
55pub const MAYBE_QUIC_WINDOW: usize = 12;
56/// Fraction of the first [`MAYBE_QUIC_WINDOW`] payload-bearing packets that
57/// must look like QUIC headers for the connection to be accepted.
58/// Mid-stream QUIC is very nearly all short headers, so this is set high.
59/// The slack tolerates one stray packet.
60pub const MAYBE_QUIC_MIN_FRACTION: f64 = 0.9;
61/// Minimum payload-bearing packets needed to judge a connection that ended
62/// before [`MAYBE_QUIC_WINDOW`] was reached. Shorter connections carry too
63/// little evidence to classify and are dropped.
64///
65/// This is somewhat arbitrarily chosen, and it will drop some genuine
66/// QUIC traffic. This has a 1/170 chance of matching random-looking traffic,
67/// but will miss some short QUIC connections.
68pub const MAYBE_QUIC_MIN_PKTS: usize = 11;
69/// 443 (HTTP/3), 853 (DNS-over-QUIC, RFC 9250), 4433 (QUIC interop/test port), 8
70/// 443 (alt HTTPS/HTTP-3). A connection on any other port is dropped without inspection.
71pub const QUIC_PORTS: [u16; 4] = [443, 853, 4433, 8443];
72/// Smallest datagram that can carry a 1-RTT packet: one header byte, a
73/// zero-length destination connection ID, and the four packet-number bytes plus
74/// 16-byte sample that RFC 9000 Section 5.4.2 requires header protection to be
75/// able to sample. Used to keep short non-QUIC datagrams out of the greased
76/// [`Shape::GreasedShortHeader`] bucket, where the fixed bit is no help.
77pub const MIN_1RTT_DATAGRAM_LEN: usize = 21;
78/// Distinct values of a matched packet's low five header bits (`first &
79/// 0x1f`) required before those packets are credited -- see
80/// [`MaybeQuic::header_entropy_ok`]. Header protection (RFC 9000 Section
81/// 5.4.1) makes those bits pseudorandom per packet.
82/// This exists to reject protocols that pin their low bits.
83/// Five is the minimum to avoid matching on OpenVPN and TURN.
84pub const MAYBE_QUIC_MIN_DISTINCT_LOW5: usize = 5;
85
86/// Number of matching packets required within a full window of
87/// [`MAYBE_QUIC_WINDOW`] packets to meet [`MAYBE_QUIC_MIN_FRACTION`], i.e.
88/// `ceil(MAYBE_QUIC_WINDOW * MAYBE_QUIC_MIN_FRACTION)`.
89pub const MAYBE_QUIC_REQUIRED_MATCHES: usize = {
90    let exact = MAYBE_QUIC_WINDOW as f64 * MAYBE_QUIC_MIN_FRACTION;
91    let truncated = exact as usize;
92    if (truncated as f64) < exact {
93        truncated + 1
94    } else {
95        truncated
96    }
97};

99/// How a datagram's first byte -- plus, for the shapes where the QUIC
100/// invariants pin a version field or a minimum size, its length -- classifies
101/// against the QUIC packet headers.
102#[derive(Debug, Clone, Copy, PartialEq, Eq)]
103enum Shape {
104    /// Short header with the fixed bit set: `01` in the top two bits.
105    ShortHeader,
106    /// Long header carrying a version Iris recognizes.
107    LongHeader,
108    /// Short *form* with the fixed bit cleared, large enough to be a 1-RTT
109    /// packet. Possible RFC 9287 greased packet. Only useful in a
110    /// connection that has also shown a [`Shape::ShortHeader`] or
111    /// [`Shape::LongHeader`] packet.
112    GreasedShortHeader,
113    /// Nothing that looks like QUIC.
114    Other,
115}
116
117/// Classifies one datagram's payload by its first byte and length.
118#[inline]
119fn classify(payload: &[u8]) -> Shape {
120    let Some(&first) = payload.first() else {
121        return Shape::Other;
122    };
123    if first & 0x80 != 0 {
124        // Long header: bytes 1..5 are version
125        if payload.len() < 5 {
126            return Shape::Other;
127        }
128        let version = u32::from_be_bytes([payload[1], payload[2], payload[3], payload[4]]);
129        // Version 0 is Version Negotiation, but "high bit set
130        // followed by four zero bytes" is a pattern that can easily be hit accidentally.
131        if version != 0 && is_quic_version(version) {
132            Shape::LongHeader
133        } else {
134            Shape::Other
135        }
136    } else if first & 0x40 != 0 {
137        Shape::ShortHeader
138    } else if payload.len() >= MIN_1RTT_DATAGRAM_LEN {
139        // First bit clear, long enough to be 1-RTT packet
140        Shape::GreasedShortHeader
141    } else {
142        Shape::Other
143    }
144}

146/// Accepts a connection once at least [`MAYBE_QUIC_MIN_FRACTION`] of its
147/// first [`MAYBE_QUIC_WINDOW`] payload-bearing packets (UDP, on one of
148/// [`QUIC_PORTS`]) look like QUIC packet headers, and those matches carry
149/// enough header entropy -- see [`MaybeQuic::header_entropy_ok`]. Non-UDP or
150/// non-QUIC-port connections are dropped immediately.
151#[cfg_attr(not(feature = "skip_expand"), filter)]
152#[derive(Debug)]
153pub struct MaybeQuic {
154    /// Payload-bearing packets inspected so far.
155    seen: usize,
156    /// Of those, how many were unambiguously QUIC-shaped: a short header with
157    /// the fixed bit set, or a long header with a recognized version.
158    quic_like: usize,
159    /// Of those, how many were fixed-bit-clear short-form datagrams big enough
160    /// to be greased 1-RTT packets. Credited only if `quic_like` is nonzero.
161    greased: usize,
162    /// Bit `v` is set once some short-form matched packet (`ShortHeader` or
163    /// `GreasedShortHeader`) had `first & 0x1f == v`. RFC 9287 greasing never
164    /// touches these bits, so greased packets contribute here even before
165    /// `quic_like` is nonzero -- see [`MaybeQuic::header_entropy_ok`].
166    low5_seen: u32,
167    /// True once a `LongHeader` packet has been seen. A 32-bit version match
168    /// is stronger evidence than the entropy check, so it satisfies that
169    /// check outright.
170    saw_long_header: bool,
171}
172
173impl StreamingFilter for MaybeQuic {
174    fn new(_first_packet: &L4Pdu) -> Self {
175        Self {
176            seen: 0,
177            quic_like: 0,
178            greased: 0,
179            low5_seen: 0,
180            saw_long_header: false,
181        }
182    }
183
184    fn clear(&mut self) {
185        self.seen = 0;
186        self.quic_like = 0;
187        self.greased = 0;
188        self.low5_seen = 0;
189        self.saw_long_header = false;
190    }
191}
192
193impl MaybeQuic {
194    /// Updates counts from one payload
195    fn record(&mut self, data: &[u8]) {
196        self.seen += 1;
197        match classify(data) {
198            Shape::ShortHeader => {
199                self.quic_like += 1;
200                self.low5_seen |= 1 << (data[0] & 0x1f);
201            }
202            Shape::LongHeader => {
203                self.quic_like += 1;
204                self.saw_long_header = true;
205            }
206            Shape::GreasedShortHeader => {
207                self.greased += 1;
208                self.low5_seen |= 1 << (data[0] & 0x1f);
209            }
210            Shape::Other => {}
211        }
212    }
213
214    /// Packets credited as QUIC. Greased packets count only once the connection
215    /// has independently shown a real QUIC packet -- see
216    /// [`Shape::GreasedShortHeader`].
217    #[inline]
218    fn matches(&self) -> usize {
219        if self.quic_like == 0 {
220            0
221        } else {
222            self.quic_like + self.greased
223        }
224    }
225
226    /// Upper bound on [`MaybeQuic::matches`] if the rest of the connection
227    /// cooperates.
228    #[inline]
229    fn reachable_matches(&self) -> usize {
230        self.quic_like + self.greased
231    }
232
233    /// True once the matched packets carry enough header entropy to trust:
234    /// [`MAYBE_QUIC_MIN_DISTINCT_LOW5`] distinct low-5-bit values, or a
235    /// recognized long header on its own.
236    #[inline]
237    fn header_entropy_ok(&self) -> bool {
238        self.saw_long_header || self.low5_seen.count_ones() as usize >= MAYBE_QUIC_MIN_DISTINCT_LOW5
239    }
240
241    /// Resolves the current counts to a `FilterResult`
242    fn decide(&self) -> FilterResult {
243        if self.matches() >= MAYBE_QUIC_REQUIRED_MATCHES && self.header_entropy_ok() {
244            return FilterResult::Accept;
245        }
246        if self.seen >= MAYBE_QUIC_WINDOW {
247            // Window exhausted
248            return FilterResult::Drop;
249        }
250        // Impossible to reach the required number of matches
251        let remaining = MAYBE_QUIC_WINDOW - self.seen;
252        if self.reachable_matches() + remaining < MAYBE_QUIC_REQUIRED_MATCHES {
253            return FilterResult::Drop;
254        }
255        // Not enough evidence yet
256        FilterResult::Continue
257    }
258
259    #[cfg_attr(not(feature = "skip_expand"), filter_fn("MaybeQuic,level=InL4Conn"))]
260    pub fn update(&mut self, pdu: &L4Pdu) -> FilterResult {
261        // Drop traffic that isn't UDP on a port QUIC is expected on
262        if pdu.ctxt.proto != UDP_PROTOCOL
263            || !(QUIC_PORTS.contains(&pdu.ctxt.src.port())
264                || QUIC_PORTS.contains(&pdu.ctxt.dst.port()))
265        {
266            return FilterResult::Drop;
267        }
268        // Skip packets with no payload
269        if pdu.length() == 0 {
270            return FilterResult::Continue;
271        }
272        // Extract data
273        if let Ok(data) = pdu.mbuf_ref().get_data_slice(pdu.offset(), pdu.length()) {
274            self.record(data);
275        }
276        self.decide()
277    }
278
279    /// Vetoes the connection once a parser has identified it, so that
280    /// `MaybeQuic` reports only what the parsers missed.
281    ///
282    /// `SessionProto` resolves at `L7OnDisc`, which is dispatched as soon as
283    /// discovery concludes -- on the first packet or two of a connection whose
284    /// handshake is visible, and so always before [`MaybeQuic::update`] can
285    /// reach [`MAYBE_QUIC_REQUIRED_MATCHES`] and accept. Returning `Drop` here
286    /// also deactivates the filter, so `update` stops being called for the rest
287    /// of the connection.
288    ///
289    /// Mid-stream QUIC -- the population this filter exists for -- never gets
290    /// here: the `quic` probe answers `Unsure` for a short header and never
291    /// concludes, so `L7OnDisc` is not dispatched at all and the heuristic
292    /// runs to its own verdict.
293    ///
294    /// One case still overlaps: a connection whose long-header packet arrives
295    /// after the window has already closed on an accept. There is no veto left
296    /// to apply at that point, and the accept was made on its own evidence.
297    #[cfg_attr(not(feature = "skip_expand"), filter_fn("MaybeQuic,level=L7OnDisc"))]
298    pub fn unclaimed(&self, proto: &SessionProto) -> FilterResult {
299        match proto {
300            // A parser claimed the connection; it is already reported as that
301            // protocol, and a heuristic guess on top would double-count it.
302            SessionProto::Tls
303            | SessionProto::Dns
304            | SessionProto::Http
305            | SessionProto::Quic
306            | SessionProto::Ssh => FilterResult::Drop,
307            // Nothing has claimed it: `Probing` while discovery is still
308            // running, `Null` once every registered parser has declined.
309            SessionProto::Null | SessionProto::Probing => FilterResult::Continue,
310            // `ConnParser::protocol` never yields the transport-layer variants
311            // -- they exist for the filter AST -- but they are listed rather
312            // than wildcarded so that adding an L7 parser is a compile error
313            // here instead of a silent hole in the veto.
314            SessionProto::Ipv4 | SessionProto::Ipv6 | SessionProto::Tcp | SessionProto::Udp => {
315                FilterResult::Continue
316            }
317        }
318    }
319
320    /// Reached only if the connection terminated before the window filled.
321    /// The fraction is applied to the packets actually observed, but only
322    /// once there are enough of them to be meaningful.
323    #[cfg_attr(
324        not(feature = "skip_expand"),
325        filter_fn("MaybeQuic,level=L4Terminated")
326    )]
327    pub fn terminated(&self) -> FilterResult {
328        if self.seen >= MAYBE_QUIC_MIN_PKTS
329            && self.header_entropy_ok()
330            && self.matches() as f64 >= self.seen as f64 * MAYBE_QUIC_MIN_FRACTION
331        {
332            FilterResult::Accept
333        } else {
334            FilterResult::Drop
335        }
336    }
337}

344    /// then version `0x00000001`.
345    const LONG_HDR_V1: [u8; 5] = [0xc3, 0x00, 0x00, 0x00, 0x01];
346    /// The same long header with the fixed bit greased away (RFC 9287).
347    const LONG_HDR_V1_GREASED: [u8; 5] = [0x83, 0x00, 0x00, 0x00, 0x01];
348
349    /// A fixed-bit-clear short-form datagram long enough to be a greased
350    /// 1-RTT packet.
351    fn greased(first: u8) -> [u8; MIN_1RTT_DATAGRAM_LEN] {
352        let mut buf = [0u8; MIN_1RTT_DATAGRAM_LEN];
353        buf[0] = first;
354        buf
355    }

358    fn short_header_bit_pattern() {
359        assert_eq!(classify(&[0x41, 0x00]), Shape::ShortHeader);
360        assert_eq!(classify(&[0x40]), Shape::ShortHeader);
361        assert_eq!(classify(&[0x7f]), Shape::ShortHeader);
362        // Fixed bit clear and too short to be a 1-RTT packet.
363        assert_eq!(classify(&[0x00]), Shape::Other);
364        // Long header, but no room for a version field.
365        assert_eq!(classify(&[0xc0]), Shape::Other);
366        assert_eq!(classify(&[0x80]), Shape::Other);
367        assert_eq!(classify(&[]), Shape::Other);

372        assert_eq!(classify(&LONG_HDR_V1), Shape::LongHeader);
373        // The fixed bit may be greased on a long header too; the version
374        // carries the decision.

699        let f = MaybeQuic::new_for_test();
700        // Any parser claiming the connection takes it out of `MaybeQuic`'s
701        // population -- not just the `quic` parser.
702        for proto in [
703            SessionProto::Quic,
704            SessionProto::Tls,
705            SessionProto::Dns,
706            SessionProto::Http,

716        for proto in [SessionProto::Probing, SessionProto::Null] {

718                matches!(f.unclaimed(&proto), FilterResult::Continue),
719                "{:?} should not veto",
720                proto
721            );
722        }
